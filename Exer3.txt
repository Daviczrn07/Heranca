using System.Collections.Generic;
using Heranca;

namespace Herança
{
    internal class Program
    {
        static void Main(string[] args)
        {
            Retangulo r = new Retangulo();
            r.Altura = 5;
            r.Largura = 5;

            Console.WriteLine($"Altura Quadrado: {r.Altura}");
            Console.WriteLine($"Largura Quadrado: {r.Largura}");
            Console.WriteLine($"area Do Quadrado: {r.CalcularArea():F2}");

            Circulo c = new Circulo();
            c.Raio = 5;


            Console.WriteLine($"Raio do Circulo: {c.Raio}");
            Console.WriteLine($"Area do Circulo: {c.CalcularArea():F2}");
        }
    }
}
